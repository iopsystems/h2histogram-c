cmake_minimum_required(VERSION 3.16)
project(h2histogram VERSION 0.1.0 LANGUAGES C)
option(H2_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
include(CTest)
add_library(h2histogram STATIC src/h2histogram.c)
target_compile_features(h2histogram PUBLIC c_std_11)
set_target_properties(h2histogram PROPERTIES C_EXTENSIONS OFF)
target_include_directories(h2histogram PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(NOT MSVC)
 target_compile_options(h2histogram PRIVATE -Wall -Wextra -Wpedantic -Werror)
 target_link_libraries(h2histogram PUBLIC m)
endif()
if(H2_SANITIZE AND NOT MSVC)
 target_compile_options(h2histogram PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(h2histogram PUBLIC -fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
 add_executable(h2_tests tests/test.c)
 target_link_libraries(h2_tests PRIVATE h2histogram)
 target_compile_definitions(h2_tests PRIVATE H2_FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/geometry.csv")
 if(MSVC)
  target_compile_options(h2_tests PRIVATE /UNDEBUG)
 else()
  target_compile_options(h2_tests PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
 endif()
 add_test(NAME invariants COMMAND h2_tests)
 if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_executable(h2_allocation_tests tests/allocation.c)
  target_link_libraries(h2_allocation_tests PRIVATE h2histogram)
  target_compile_options(h2_allocation_tests PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
  target_link_options(h2_allocation_tests PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=free)
  add_test(NAME allocation COMMAND h2_allocation_tests)
 endif()
endif()
if(NOT MSVC)
 # Bounded model checking with CBMC 6.x. Proofs run sequentially; see verify/README.md.
 add_custom_target(verify COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/verify/run.sh USES_TERMINAL)
endif()
add_executable(h2_example examples/basic.c)
target_link_libraries(h2_example PRIVATE h2histogram)
install(TARGETS h2histogram ARCHIVE DESTINATION lib)
install(FILES include/h2histogram.h DESTINATION include)
